/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  /* Đặt Flash chờ 2 chu kỳ và bật prefetch trước khi tăng xung lên 72 MHz. */
  FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY) |
               FLASH_ACR_PRFTBE |
               FLASH_ACR_LATENCY_2;

  /* Bật bộ dao động ngoài HSE (thạch anh 8 MHz trên board). */
  RCC->CR |= RCC_CR_HSEON;
  /* Chờ HSE ổn định trước khi dùng làm nguồn cho PLL. */
  while ((RCC->CR & RCC_CR_HSERDY) == 0U)
  {
  }

  /* Xóa cấu hình clock cũ: nguồn SYSCLK, prescaler bus và cấu hình PLL. */
  RCC->CFGR = (RCC->CFGR & ~(RCC_CFGR_SW |
                             RCC_CFGR_HPRE |
                             RCC_CFGR_PPRE1 |
                             RCC_CFGR_PPRE2 |
                             RCC_CFGR_ADCPRE |
                             RCC_CFGR_PLLSRC |
                             RCC_CFGR_PLLXTPRE |
                             RCC_CFGR_PLLMULL)) |
              /* APB1 chia 2: 72 MHz / 2 = 36 MHz, đúng giới hạn STM32F1. */
              RCC_CFGR_PPRE1_DIV2 |
              /* ADC clock chia PCLK2 cho 6: 72 MHz / 6 = 12 MHz. */
              RCC_CFGR_ADCPRE_DIV6 |
              /* Chọn HSE làm đầu vào PLL, không chia HSE trước PLL. */
              RCC_CFGR_PLLSRC |
              /* Nhân PLL lên 9: HSE 8 MHz x 9 = SYSCLK 72 MHz. */
              RCC_CFGR_PLLMULL9;

  /* Bật PLL sau khi đã thiết lập nguồn và hệ số nhân. */
  RCC->CR |= RCC_CR_PLLON;
  /* Chờ PLL khóa xong trước khi chuyển SYSCLK sang PLL. */
  while ((RCC->CR & RCC_CR_PLLRDY) == 0U)
  {
  }

  /* Chọn PLL làm nguồn xung hệ thống. */
  RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
  /* Chờ phần cứng xác nhận PLL đang cấp xung hệ thống. */
  while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL)
  {
  }

  /* Cập nhật biến CMSIS SystemCoreClock để các phần khác biết CPU chạy 72 MHz. */
  SystemCoreClockUpdate();
}
